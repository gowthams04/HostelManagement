📘 How to set up your Google Apps Script backend
  1. Create a new Google Apps Script project at script.google.com.
  2. Paste the backend code (see below) into Code.gs.
  3. Click Deploy → New deployment → Web app.
  4. Set Execute as → Me and Who has access → Anyone (or your domain).
  5. Click Deploy and copy the URL (must end with /exec).
  6. Create a Google Sheet with sheets named: Hostels, Rooms, Students, GuestBookings.
  7. Add column headers exactly as shown in the examples below.
  8. Paste the Web App URL and Sheet ID here, then click Save Settings and Test Connection.

Sample sheet structure:

  • Hostels: id, name, type, floors, chiefWarden, phone, status
  • Rooms: id, hostelId, floorId, number, category, type, capacity, bathroom, ac, status
  • Students: id, admissionNo, rollNo, name, gender, department, year, phone, parentPhone, email, bloodGroup, address, status, hostelId, floorId, roomId
  • GuestBookings: id, facultyName, department, guestName, purpose, hostelId, floorId, roomId, checkIn, checkOut, status

Backend code (Code.gs):

function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  var func = data.func;
  var spreadsheetId = data.spreadsheetId;
  try {
    var result;
    switch(func) {
      case 'testConnection': result = { success: true }; break;
      case 'getDashboardStats': result = getDashboardStats(spreadsheetId); break;
      case 'getHostels': result = getSheetData(spreadsheetId, 'Hostels'); break;
      case 'getRooms': result = getSheetData(spreadsheetId, 'Rooms'); break;
      case 'getStudents': result = getSheetData(spreadsheetId, 'Students'); break;
      default: throw new Error('Unknown function');
    }
    return ContentService.createTextOutput(JSON.stringify({ result: result }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({ error: err.message }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
function getSheetData(id, name) {
  var sheet = SpreadsheetApp.openById(id).getSheetByName(name);
  if(!sheet) return [];
  var data = sheet.getDataRange().getValues();
  var headers = data.shift();
  return data.map(function(row) {
    var obj = {};
    headers.forEach(function(h,i) { obj[h] = row[i]; });
    return obj;
  });
}
function getDashboardStats(id) {
  return { totalHostels: getSheetData(id,'Hostels').length, totalRooms: getSheetData(id,'Rooms').length, totalStudents: getSheetData(id,'Students').length, totalGuestBookings: getSheetData(id,'GuestBookings').length };
}
